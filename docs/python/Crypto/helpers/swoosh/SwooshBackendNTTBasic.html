 19class PolyNTT:
 20    """
 21    Representación mínima de un polinomio para el backend.
 22    Internamente almacena la lista de coeficientes en el dominio habitual
 23    de coeficientes, no en el dominio transformado.
 24    """
 25
 26    __slots__ = ("_c",)
 27
 28    def __init__(self, coeffs: List[int]):
 29        """
 30        Inicializa el polinomio a partir de una lista de coeficientes.
 31        """
 32        self._c = coeffs
 33
 34    def coeffs(self) -> List[int]:
 35        """
 36        Devuelve una copia de los coeficientes del polinomio.
 37        """
 38        return list(self._c)
 39
 40    def __eq__(self, other: object) -> bool:
 41        """
 42        Compara dos polinomios por igualdad de coeficientes.
 43        """
 44        if not isinstance(other, PolyNTT):
 45            return NotImplemented
 46        return self._c == other._c
 47
 48    def __repr__(self) -> str:
 49        """
 50        Devuelve una representación abreviada del polinomio.
 51        """
 52        if not self._c:
 53            return "PolyNTT([])"
 54        h = hashlib.sha256(
 55            b"".join(
 56                int(x).to_bytes(4, "little", signed=False)
 57                for x in self._c[:32]
 58            )
 59        ).hexdigest()[:12]
 60        return f"PolyNTT(d={len(self._c)}, fp={h})"

 63class BackendNTTBasic(SwooshBackend):
 64    """
 65    Backend NTT/FTT para el anillo R_q = Z_q[x] / (x^d + 1),
 66    con d potencia de 2.
 67    """
 68
 69    def __init__(self, params: SwooshParameters):
 70        """
 71        Inicializa el backend y construye el contexto NTT asociado.
 72        Valida que el módulo elegido sea compatible con la
 73        condición negacíclica necesaria para la transformada.
 74        """
 75        self.p = params
 76        self.d = int(params.d)
 77        self.N = int(params.N)
 78        self.q = int(params.q_ntt)
 79        self.elem_bytes = int(
 80            getattr(params, "elem_bytes", (self.q.bit_length() + 7) // 8)
 81        )
 82
 83        # Condición NTT negacíclica: 2d | (q - 1)
 84        if (self.q - 1) % (2 * self.d) != 0:
 85            raise ValueError(
 86                "NTTBasicBackend: se requiere q ≡ 1 (mod 2d) "
 87                "para FTT/NTT negacíclica."
 88            )
 89
 90        self.ntt_ctx = NTTContext(
 91            poly_degree=self.d,
 92            coeff_modulus=self.q,
 93        )
 94
 95        # rejection sampling para uniformes
 96        k = 8 * self.elem_bytes
 97        self.M = 1 << k
 98        self.T = self.M - (self.M % self.q)

120    def add_rq(self, a: PolyNTT, b: PolyNTT) -> PolyNTT:
121        """
122        Suma dos polinomios en R_q coeficiente a coeficiente.
123        """
124        return PolyNTT(
125            [(a._c[i] + b._c[i]) % self.q for i in range(self.d)]
126        )
127
128    def mul_rq(self, a: PolyNTT, b: PolyNTT) -> PolyNTT:
129        """
130        Multiplica dos polinomios mediante FTT/NTT negacíclica.
131        """
132        A = self.ntt_ctx.ftt_fwd(a._c)
133        B = self.ntt_ctx.ftt_fwd(b._c)
134        C = [(A[i] * B[i]) % self.q for i in range(self.d)]
135        c = self.ntt_ctx.ftt_inv(C)
136        return PolyNTT(c)

138    def _zero_poly(self) -> PolyNTT:
139        """
140        Devuelve el polinomio nulo del anillo.
141        """
142        return PolyNTT([0] * self.d)

144    def _zero_vec(self) -> Vec:
145        """
146        Construye un vector nulo de longitud N.
147        """
148        return [self._zero_poly() for _ in range(self.N)]

150    def _matvec(self, A: Mat, v: Vec) -> Vec:
151        """
152        Calcula el producto matriz-vector A · v en el caso N > 1.
153        """
154        out = self._zero_vec()
155        for i in range(self.N):
156            acc = self._zero_poly()
157            for j in range(self.N):
158                acc = self.add_rq(acc, self.mul_rq(A[i][j], v[j]))
159            out[i] = acc
160        return out
161
162    def _vecmat(self, s: Vec, A: Mat) -> Vec:
163        """
164        Calcula el producto vector-matriz s^T · A en el caso N > 1.
165        """
166        out = self._zero_vec()
167        for j in range(self.N):
168            acc = self._zero_poly()
169            for i in range(self.N):
170                acc = self.add_rq(acc, self.mul_rq(s[i], A[i][j]))
171            out[j] = acc
172        return out
173
174    def _dot(self, u: Vec, v: Vec) -> PolyNTT:
175        """
176        Calcula el producto escalar de dos vectores de polinomios en R_q.
177        """
178        acc = self._zero_poly()
179        for i in range(self.N):
180            acc = self.add_rq(acc, self.mul_rq(u[i], v[i]))
181        return acc

186    def poly_to_bytes(self, p: PolyNTT) -> bytes:
187        """
188        Serializa un polinomio a bytes en formato little-endian.
189        """
190        out = bytearray(self.d * self.elem_bytes)
191        for i in range(self.d):
192            out[
193                i * self.elem_bytes:(i + 1) * self.elem_bytes
194            ] = int(p._c[i]).to_bytes(self.elem_bytes, "little")
195        return bytes(out)

197    def poly_from_bytes(self, buf: bytes) -> PolyNTT:
198        """
199        Reconstruye un polinomio a partir de su representación en bytes.
200        """
201        expected = self.d * self.elem_bytes
202        if len(buf) != expected:
203            raise ValueError(
204                f"poly_from_bytes: esperaba {expected} bytes y recibió {len(buf)}"
205            )
206
207        coeffs = []
208        for i in range(self.d):
209            chunk = buf[i * self.elem_bytes:(i + 1) * self.elem_bytes]
210            coeffs.append(int.from_bytes(chunk, "little") % self.q)
211
212        return PolyNTT(coeffs)

214    def _vec_to_bytes(self, v: Vec) -> bytes:
215        """
216        Serializa un vector de polinomios concatenando sus elementos.
217        """
218        return b"".join(self.poly_to_bytes(p) for p in v)

220    def _vec_from_bytes(self, buf: bytes) -> Vec:
221        """
222        Reconstruye un vector de polinomios a partir de bytes.
223        """
224        poly_bytes = self.d * self.elem_bytes
225        expected = self.N * poly_bytes
226
227        if len(buf) != expected:
228            raise ValueError(
229                f"_vec_from_bytes: esperaba {expected} bytes y recibió {len(buf)}"
230            )
231
232        return [
233            self.poly_from_bytes(buf[i * poly_bytes:(i + 1) * poly_bytes])
234            for i in range(self.N)
235        ]

237    def split_lr_bytes(self, buf: bytes) -> Tuple[bytes, bytes]:
238        """
239        Divide una representación L||R en sus mitades izquierda y derecha.
240        """
241        poly_bytes = self.d * self.elem_bytes
242        side_bytes = poly_bytes if self.N == 1 else self.N * poly_bytes
243
244        if len(buf) != 2 * side_bytes:
245            raise ValueError(
246                f"split_lr_bytes: esperaba {2 * side_bytes} bytes y recibió {len(buf)}"
247            )
248
249        return buf[:side_bytes], buf[side_bytes:]
250
251    def join_lr_bytes(self, left: bytes, right: bytes) -> bytes:
252        """
253        Une las mitades izquierda y derecha en una sola secuencia binaria.
254        """
255        return left + right

257    def serialize_sk(self, sk: Sk) -> bytes:
258        """
259        Serializa la clave secreta del backend en formato L||R.
260        """
261        skL, skR = sk
262        left = self.poly_to_bytes(skL) if self.N == 1 else self._vec_to_bytes(skL)
263        right = self.poly_to_bytes(skR) if self.N == 1 else self._vec_to_bytes(skR)
264        return self.join_lr_bytes(left, right)

266    def deserialize_sk(self, buf: bytes) -> Sk:
267        """
268        Reconstruye la clave secreta a partir de su representación serializada.
269        """
270        Lb, Rb = self.split_lr_bytes(buf)
271        left = self.poly_from_bytes(Lb) if self.N == 1 else self._vec_from_bytes(Lb)
272        right = self.poly_from_bytes(Rb) if self.N == 1 else self._vec_from_bytes(Rb)
273        return (left, right)

276        """
277        Serializa la clave pública del backend en formato L||R.

279        pkL, pkR = pk
280        left = self.poly_to_bytes(pkL) if self.N == 1 else self._vec_to_bytes(pkL)
281        right = self.poly_to_bytes(pkR) if self.N == 1 else self._vec_to_bytes(pkR)
282        return self.join_lr_bytes(left, right)

284    def deserialize_pk(self, buf: bytes) -> Pk:
285        """
286        Reconstruye la clave pública a partir de su representación serializada.
287        """
288        Lb, Rb = self.split_lr_bytes(buf)

290        right = self.poly_from_bytes(Rb) if self.N == 1 else self._vec_from_bytes(Rb)

296    def setup_A(self, seed_A: bytes) -> Union[PolyNTT, Mat]:
297        """
298        Construye la estructura pública A a partir de la semilla pública.
299        Para N=1 devuelve un polinomio; para N>1 construye una matriz
300        N x N de polinomios compatible con el backend NTT.
301        """
302        if self.N == 1:
303            coeffs = []
304            ctr = 0
305            while len(coeffs) < self.d:
306                block = self._shake(
307                    self.p.domain_base,
308                    self.p.domain_A,
309                    seed_A,
310                    ctr.to_bytes(4, "little"),
311                    outlen=self.elem_bytes,
312                )
313                ctr += 1
314                u = int.from_bytes(block, "little")
315                if u < self.T:
316                    coeffs.append(u % self.q)
317            return PolyNTT(coeffs)
318
319        A: Mat = [[self._zero_poly() for _ in range(self.N)] for _ in range(self.N)]
320        ctr = 0
321        for i in range(self.N):
322            for j in range(self.N):
323                coeffs = []
324                while len(coeffs) < self.d:
325                    block = self._shake(
326                        self.p.domain_base,
327                        self.p.domain_A,
328                        seed_A,
329                        i.to_bytes(2, "little"),
330                        j.to_bytes(2, "little"),
331                        ctr.to_bytes(4, "little"),
332                        outlen=self.elem_bytes,
333                    )
334                    ctr += 1
335                    u = int.from_bytes(block, "little")
336                    if u < self.T:
337                        coeffs.append(u % self.q)
338                A[i][j] = PolyNTT(coeffs)
339        return A

341    def keygen(self, A: Union[PolyNTT, Mat]) -> Tuple[Sk, Pk]:
342        """
343        Genera la clave secreta y la clave pública.
344        """
345        noiseseed = secrets.token_bytes(32)
346
347        if self.N == 1:
348            skL = self.sample_cbd_poly(self.p.domain_sk, noiseseed, nonce=0, eta=1)
349            skR = self.sample_cbd_poly(self.p.domain_sk, noiseseed, nonce=1, eta=1)
350            eL = self.sample_cbd_poly(self.p.domain_e, noiseseed, nonce=2, eta=1)
351            eR = self.sample_cbd_poly(self.p.domain_e, noiseseed, nonce=3, eta=1)
352
353            pkL = self.add_rq(self.mul_rq(skL, A), eL)   # s^T A (trivial N=1)
354            pkR = self.add_rq(self.mul_rq(A, skR), eR)   # A s
355            return (skL, skR), (pkL, pkR)
356
357        # N > 1 
358        skL: Vec = [
359            self.sample_cbd_poly(self.p.domain_sk, noiseseed, nonce=10 + i, eta=1)
360            for i in range(self.N)
361        ]
362        skR: Vec = [
363            self.sample_cbd_poly(self.p.domain_sk, noiseseed, nonce=20 + i, eta=1)
364            for i in range(self.N)
365        ]
366        eL: Vec = [
367            self.sample_cbd_poly(self.p.domain_e, noiseseed, nonce=30 + i, eta=1)
368            for i in range(self.N)
369        ]
370        eR: Vec = [
371            self.sample_cbd_poly(self.p.domain_e, noiseseed, nonce=40 + i, eta=1)
372            for i in range(self.N)
373        ]
374
375        pkL_vec = self._vecmat(skL, A)     # sL^T A
376        pkR_vec = self._matvec(A, skR)     # A sR
377
378        pkL = [self.add_rq(pkL_vec[i], eL[i]) for i in range(self.N)]
379        pkR = [self.add_rq(pkR_vec[i], eR[i]) for i in range(self.N)]
380        return (skL, skR), (pkL, pkR)

386    def derive_shared_key(
387        self,
388        my_sk: bytes,
389        my_pk: bytes,
390        peer_pk: bytes,
391    ) -> bytes:
392        """
393        Deriva la shared key a partir del material local y de la clave pública del peer.
394        1. fijar el orden canónico de claves públicas
395        2. construir el offset r
396        3. deserializar clave pública y secreta
397        4. calcular k'
398        5. reconciliar el resultado para obtener la calve compartida
399        """
400        pk1, pk2 = self.cardinality_by_bytes(my_pk, peer_pk)
401        r = self.offset_poly(self.p.domain_offset, pk1, pk2)
402
403        pk_peer = self.deserialize_pk(peer_pk)
404        sk_me = self.deserialize_sk(my_sk)
405
406        kprime = self.kprime(pk_peer, sk_me, r)
407        self.shared_key = self.reconcile(kprime)
408        return self.shared_key

410    def offset_poly(self, label: bytes, pk1: bytes, pk2: bytes) -> PolyNTT:
411        """
412        Construye el polinomio de offset a partir de las claves públicas ordenadas.
413        """
414        rin = pk1 + pk2
415        coeffs = []
416
417        for i in range(self.d):
418            ctr = 0
419            while True:
420                block = self._shake(
421                    self.p.domain_base,
422                    label,
423                    rin,
424                    i.to_bytes(4, "little"),
425                    ctr.to_bytes(4, "little"),
426                    outlen=self.elem_bytes,
427                )
428                u = int.from_bytes(block, "little")
429                if u < self.T:
430                    coeffs.append(u % self.q)
431                    break
432                ctr += 1
433
434        return PolyNTT(coeffs)

436    def kprime(self, peer_pk: Pk, my_sk: Sk, r: PolyNTT) -> PolyNTT:
437        """
438        Calcula el valor preliminar k' antes de la reconciliación final.
439        """
440        pkL_peer, pkR_peer = peer_pk
441        skL_me, skR_me = my_sk
442
443        if self.N == 1:
444            term1 = self.mul_rq(skL_me, pkR_peer)      # sL * pkR
445            term2 = self.mul_rq(pkL_peer, skR_me)      # pkL * sR
446            return self.add_rq(self.add_rq(term1, term2), r)
447
448        term1 = self._dot(skL_me, pkR_peer)            # <sL, pkR>
449        term2 = self._dot(pkL_peer, skR_me)            # <pkL, sR>
450        return self.add_rq(self.add_rq(term1, term2), r)

452    def reconcile(self, kv_poly: Poly) -> bytes:
453        """
454        Reconciliación final de k' para obtener una secuencia binaria.
455        Cada coeficiente se proyecta a un bit según su posición relativa
456        respecto a los umbrales q/4 y 3q/4.
457        """
458        coeffs = list(kv_poly.coeffs())
459        if len(coeffs) < self.d:
460            coeffs += [0] * (self.d - len(coeffs))
461
462        q = int(self.q)
463        out = bytearray(self.d // 8)
464
465        q4 = q // 4
466        tq4 = (3 * q) // 4
467
468        for i in range(self.d // 8):
469            acc = 0
470            for j in range(8):
471                c = int(coeffs[8 * i + j]) % q
472                bit = 1 if q4 <= c <= tq4 else 0
473                acc |= bit << j
474            out[i] = acc
475
476        return bytes(out[: self.p.key_bytes])

132        """
133        A = self.ntt_ctx.ftt_fwd(a._c)

342    def keygen(self, A: Union[PolyNTT, Mat]) -> Tuple[Sk, Pk]:
343        """
344        Genera la clave secreta y la clave pública.
345        """
346        noiseseed = secrets.token_bytes(32)
347
348        if self.N == 1:
349            skL = self.sample_cbd_poly(self.p.domain_sk, noiseseed, nonce=0, eta=1)
350            skR = self.sample_cbd_poly(self.p.domain_sk, noiseseed, nonce=1, eta=1)
351            eL = self.sample_cbd_poly(self.p.domain_e, noiseseed, nonce=2, eta=1)
352            eR = self.sample_cbd_poly(self.p.domain_e, noiseseed, nonce=3, eta=1)

354            pkL = self.add_rq(self.mul_rq(skL, A), eL)   # s^T A (trivial N=1)
355            pkR = self.add_rq(self.mul_rq(A, skR), eR)   # A s
356            return (skL, skR), (pkL, pkR)
357
358        # N > 1 
359        skL: Vec = [
360            self.sample_cbd_poly(self.p.domain_sk, noiseseed, nonce=10 + i, eta=1)
361            for i in range(self.N)
362        ]
363        skR: Vec = [
364            self.sample_cbd_poly(self.p.domain_sk, noiseseed, nonce=20 + i, eta=1)
365            for i in range(self.N)
366        ]
367        eL: Vec = [
368            self.sample_cbd_poly(self.p.domain_e, noiseseed, nonce=30 + i, eta=1)
369            for i in range(self.N)
370        ]
371        eR: Vec = [
372            self.sample_cbd_poly(self.p.domain_e, noiseseed, nonce=40 + i, eta=1)
373            for i in range(self.N)
374        ]
375
376        pkL_vec = self._vecmat(skL, A)     # sL^T A
377        pkR_vec = self._matvec(A, skR)     # A sR
378
379        pkL = [self.add_rq(pkL_vec[i], eL[i]) for i in range(self.N)]
380        pkR = [self.add_rq(pkR_vec[i], eR[i]) for i in range(self.N)]
381        return (skL, skR), (pkL, pkR)

387    def derive_shared_key(
388        self,
389        my_sk: bytes,
390        my_pk: bytes,
391        peer_pk: bytes,
392    ) -> bytes:
393        """
394        Deriva la shared key a partir del material local y de la clave pública del peer.
395        1. fijar el orden canónico de claves públicas
396        2. construir el offset r
397        3. deserializar clave pública y secreta
398        4. calcular k'
399        5. reconciliar el resultado para obtener la calve compartida
400        """
401        pk1, pk2 = self.cardinality_by_bytes(my_pk, peer_pk)
402        r = self.offset_poly(self.p.domain_offset, pk1, pk2)
403
404        pk_peer = self.deserialize_pk(peer_pk)
405        sk_me = self.deserialize_sk(my_sk)
406
407        kprime = self.kprime(pk_peer, sk_me, r)
408        self.shared_key = self.reconcile(kprime)
409        return self.shared_key

411    def offset_poly(self, label: bytes, pk1: bytes, pk2: bytes) -> PolyNTT:
412        """
413        Construye el polinomio de offset a partir de las claves públicas ordenadas.
414        """
415        rin = pk1 + pk2
416        coeffs = []
417
418        for i in range(self.d):
419            ctr = 0

421                block = self._shake(

428                )

437    def kprime(self, peer_pk: Pk, my_sk: Sk, r: PolyNTT) -> PolyNTT:
438        """
439        Calcula el valor preliminar k' antes de la reconciliación final.
440        """
441        pkL_peer, pkR_peer = peer_pk
442        skL_me, skR_me = my_sk
443
444        if self.N == 1:
445            term1 = self.mul_rq(skL_me, pkR_peer)      # sL * pkR
446            term2 = self.mul_rq(pkL_peer, skR_me)      # pkL * sR
447            return self.add_rq(self.add_rq(term1, term2), r)
448
449        term1 = self._dot(skL_me, pkR_peer)            # <sL, pkR>
450        term2 = self._dot(pkL_peer, skR_me)            # <pkL, sR>
451        return self.add_rq(self.add_rq(term1, term2), r)

453    def reconcile(self, kv_poly: Poly) -> bytes:
454        """
455        Reconciliación final de k' para obtener una secuencia binaria.
456        Cada coeficiente se proyecta a un bit según su posición relativa
457        respecto a los umbrales q/4 y 3q/4.
458        """
459        coeffs = list(kv_poly.coeffs())
460        if len(coeffs) < self.d:
461            coeffs += [0] * (self.d - len(coeffs))
462
463        q = int(self.q)
464        out = bytearray(self.d // 8)
465
466        q4 = q // 4
467        tq4 = (3 * q) // 4
468
469        for i in range(self.d // 8):
470            acc = 0
471            for j in range(8):
472                c = int(coeffs[8 * i + j]) % q
473                bit = 1 if q4 <= c <= tq4 else 0
474                acc |= bit << j
475            out[i] = acc
476
477        return bytes(out[: self.p.key_bytes])

def reconcile(self, kv_poly: Any) -> bytes: 
453    def reconcile(self, kv_poly: Poly) -> bytes:
454        """
455        Reconciliación final de k' para obtener una secuencia binaria.
456        Cada coeficiente se proyecta a un bit según su posición relativa
457        respecto a los umbrales q/4 y 3q/4.
458        """
459        coeffs = list(kv_poly.coeffs())
460        if len(coeffs) < self.d:
461            coeffs += [0] * (self.d - len(coeffs))
462
463        q = int(self.q)
464        out = bytearray(self.d // 8)
465
466        q4 = q // 4
467        tq4 = (3 * q) // 4
468
469        for i in range(self.d // 8):
470            acc = 0
471            for j in range(8):
472                c = int(coeffs[8 * i + j]) % q
473                bit = 1 if q4 <= c <= tq4 else 0
474                acc |= bit << j
475            out[i] = acc
476
477        return bytes(out[: self.p.key_bytes])
